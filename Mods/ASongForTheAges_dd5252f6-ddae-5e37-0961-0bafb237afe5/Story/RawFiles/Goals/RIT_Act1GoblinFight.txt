// Trigger combat when player enters the ambush zone
IF
EnteredTrigger(_Player, (TRIGGER)RIT_GoblinFight_Act1_EventTrigger_7948f6aa-976c-485c-890b-1d90b1d3c94b)
AND
DB_Players(_Player)
AND
NOT DB_SFAGoblinFightStarted(1)
THEN
DB_SFAGoblinFightStarted(1);
QuestUpdate("RIT_ASongForTheAges", "RIT_KillGoblins_Step");
PROC_RIT_StartGoblinFight(_Player);

// Set each goblin's faction and force them into combat against the player
PROC
PROC_RIT_StartGoblinFight((CHARACTER)_Player)
AND
DB_SFAGoblins(_Goblin)
AND
GetFaction(_Player, _PlayerFaction)
THEN
SetIndividualRelation(_Goblin, _PlayerFaction, 0);
EnterCombat(_Goblin, _Player);

// Remove non-last goblins from tracking on death
IF
KilledBy(_Goblin, _AttackOwner, _Attacker, _StoryActionID)
AND
DB_SFAGoblins(_Goblin)
AND
SysCount("DB_SFAGoblins", 1, _Count)
AND
_Count > 1
THEN
NOT DB_SFAGoblins(_Goblin);

// Last goblin dead: advance quest and unlock the marker pointing to Dafodil
IF
KilledBy(_Goblin, _AttackOwner, _Attacker, _StoryActionID)
AND
DB_SFAGoblins(_Goblin)
AND
SysCount("DB_SFAGoblins", 1, _Count)
AND
_Count == 1
THEN
NOT DB_SFAGoblins(_Goblin);
QuestUpdate("RIT_ASongForTheAges", "RIT_SpeakToDafodil_Step");
EXITSECTION
