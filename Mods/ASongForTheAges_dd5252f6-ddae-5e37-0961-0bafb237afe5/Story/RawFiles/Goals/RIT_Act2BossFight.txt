// Player walks into the trigger zone: start the fight
IF
EnteredTrigger(_Player, (TRIGGER)RIT_Act2BossFight_EventTrigger_ca309c5a-d26a-45ff-a83b-22867c1e7fe0)
AND
DB_Players(_Player)
AND
NOT DB_SFABossFightStarted(1)
THEN
DB_SFABossFightStarted(1);
QuestUpdate("RIT_ASongForTheAges", "RIT_FightTrexel_Step");
PROC_RIT_StartBossFight(_Player);

IF
EnteredTrigger(_Player, (TRIGGER)RIT_Act2BossFight_EventTrigger_ca309c5a-d26a-45ff-a83b-22867c1e7fe0)
AND
DB_FightEnemies(_Enemny)
AND
GetDisplayName(_Enemny, _translatedStringKey)
AND
ResolveTranslatedString(_translatedStringKey, _displayName)
THEN
DebugText(_Player, _displayName);


PROC
PROC_RIT_StartBossFight((CHARACTER)_Player)
AND
DB_FightEnemies(_Enemy)
AND
GetFaction(_Player, _PlayerFaction)
AND
GetFaction(_Enemy, _EnemyFaction)
THEN
SetHostileAndEnterCombat(_EnemyFaction, _PlayerFaction, _Enemy, _Player);


// Non-last enemy dies: remove from tracking
IF
KilledBy(_Enemy, _AttackOwner, _Attacker, _StoryActionID)
AND
DB_FightEnemies(_Enemy)
AND
SysCount("DB_FightEnemies", 1, _Count)
AND
_Count > 1
THEN
NOT DB_FightEnemies(_Enemy);


// Last enemy dies: advance quest to claim the lute
IF
KilledBy(_Enemy, _AttackOwner, _Attacker, _StoryActionID)
AND
DB_FightEnemies(_Enemy)
AND
SysCount("DB_FightEnemies", 1, _Count)
AND
_Count == 1
THEN
NOT DB_FightEnemies(_Enemy);
QuestUpdate("RIT_ASongForTheAges", "RIT_ClaimTheLute_Step");


// Player picks up the lute: advance quest to return to Dafodil
IF
TemplateAddedTo((ITEMROOT)RIT_Dafodills_Lute_e09c7e42-978f-4f54-b321-28748365d627, _LuteInst, _Holder, _AddType)
AND
DB_Players((CHARACTER)_Holder)
AND
NOT DB_SFALuteClaimed(1)
THEN
DB_SFALuteClaimed(1);
QuestUpdate("RIT_ASongForTheAges", "RIT_ReturnToBrokenstring_Step");
